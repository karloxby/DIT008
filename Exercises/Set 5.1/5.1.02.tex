		\begin{adjustwidth}{0cm}{0cm}
			Write the first four terms of the sequences defined by the
			formulas in 1–6.

			\[
				b_j = \frac{5 - j}{5 + j}\text{, for every integer j } \geq 1
			\]
		\end{adjustwidth}
	\vspace{0.25cm}
	\vspace{0.25cm}
		\begin{adjustwidth}{0cm}{0cm}
			We must show \(a_1, a_2, a_3, a_4\).

			\[
				\begin{aligned}
					a_1 = & \frac{5 - 1}{5 + 1} = \frac{4}{6} = \boxed{\frac{2}{3}} \\
					a_2 = & \frac{5 - 2}{5 + 2} = \boxed{\frac{3}{7}} \\
					a_3 = & \frac{5 - 3}{5 + 3} = \frac{2}{8} = \boxed{\frac{1}{4}} \\
					a_4 = & \frac{5 - 4}{5 + 4} = \boxed{\frac{1}{9}}
				\end{aligned}
			\]
		\end{adjustwidth}
	\vspace{0.25cm}
	\vspace{0.25cm}
		\begin{adjustwidth}{0cm}{0cm}
			\textbf{Solution:} \(a_1 = \frac{2}{3}\), \(a_2 = \frac{3}{7}\),
			\(a_3 = \frac{1}{4}\), \(a_4 = \frac{1}{9}\)
		\end{adjustwidth}
