	\definition{Definition}{
		If a sum with a variable number of terms is shown to
		equal an expression that does not contain either an ellipsis
		or a summation symbol, we say that the sum is written
		\textbf{in closed form}.
	}

	\container{
		For example, writing \(1 + 2 + 3 + \text{...} + n = \frac{n(n + 1)}{2}\) 
		expresses the sum \(1 + 2 + 3 + \text{...} + n\) in closed form.
	}
